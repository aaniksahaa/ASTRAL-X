# ASTRAL-X working commands

# Build
bash build.sh

# Core runner
./run.sh -i all_gt_bs_rooted_37.tre -o /tmp/astralx37_local.tre -vv
./run.sh -i all_gt_bs_rooted_37.tre -o /tmp/astralx37_full.tre --search-mode full -vv

# Compatibility wrapper
./run-astralx.sh -i all_gt_bs_rooted_37.tre -o /tmp/astralx37_full_via_wrapper.tre --search-mode full -vv

# Monitored standard run on 37 taxa
./run-astralx-with-monitor.sh -i all_gt_bs_rooted_37.tre -o /tmp/astralx37_monitored.tre --search-mode full -vv --no-notify --no-build

# Simulate 100 taxa, 100 gene trees, 1 replicate
./sim.sh -t 100 -g 100 --sb 0.000001 --spmin 100000 --spmax 200000 -rs 1 --fresh

# Run ASTRAL-X on that simulated replicate and write stat-astralx.csv
./test-astralx-simulated.sh -t 100 -g 100 --sb 0.000001 --spmin 100000 --spmax 200000 -r R1 --fresh

# Collect merged simulated stats
./collect-stats-simulated.sh --simphy-dir ./simphy --out /tmp/astralx-sim-stats.csv
