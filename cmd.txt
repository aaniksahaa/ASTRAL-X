# Build Java classes
bash build.sh

# ── Automated test (runs Mode 1 + Mode 2 on GPU, prints RF vs true tree) ─────
bash test_rf.sh 37          # 37-taxa dataset
bash test_rf.sh 48          # 48-taxa dataset
bash test_rf.sh 200         # 200-taxa dataset
bash test_rf.sh all         # all three datasets in sequence

# Build CUDA JNI shared libraries (once, or after changing .cu files)
bash build_native.sh

# ── 37-taxa ──────────────────────────────────────────────────────────────────

# Mode 1 (tree-local DP, GPU weight)
java -Djava.library.path=native -cp build astralx.Main -i all_gt_bs_rooted_37.tre -o astralx_species_tree_37.newick --gpu -vv

# Mode 2 (cross-tree DP, GPU weight + GPU DP search)
java -Djava.library.path=native -cp build astralx.Main -i all_gt_bs_rooted_37.tre -o astralx_species_tree_37_full.newick --gpu --search-mode full -vv

# ── 100-taxa ─────────────────────────────────────────────────────────────────

java -Djava.library.path=native -cp build astralx.Main -i all_gt_bs_rooted_100.tre -o astralx_species_tree_100.newick --gpu -vv
java -Djava.library.path=native -cp build astralx.Main -i all_gt_bs_rooted_100.tre -o astralx_species_tree_100_full.newick --gpu --search-mode full -vv

# ── 200-taxa ─────────────────────────────────────────────────────────────────

java -Djava.library.path=native -cp build astralx.Main -i all_gt_bs_rooted_200.tre -o astralx_species_tree_200.newick --gpu -vv
java -Djava.library.path=native -cp build astralx.Main -i all_gt_bs_rooted_200.tre -o astralx_species_tree_200_full.newick --gpu --search-mode full -vv

# CPU fallback (no GPU needed)
java -cp build astralx.Main -i all_gt_bs_rooted_200.tre -o astralx_species_tree_200_cpu.newick -vv
java -cp build astralx.Main -i all_gt_bs_rooted_200.tre -o astralx_species_tree_200_cpu_full.newick --search-mode full -vv
