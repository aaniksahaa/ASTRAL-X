###############################################################################
# GPU batching flags  (weight calculation phase)
###############################################################################
#
# Default behaviour (no flags): adaptive single-batch — all splits in one GPU
# launch.  Batching is only activated by one of the flags below.
#
# Flag priority (highest → lowest):
#   --no-gpu-batch  >  --gpu-batches  >  --gpu-batch-size  >  auto
#
# --no-gpu-batch
#     Disable batching entirely.  All splits sent in one kernel launch.
#     Equivalent to original single-launch behaviour.
#
# --gpu-batches N
#     Force exactly N batches regardless of available VRAM.
#     batchSize = ceil(numSplits / N) is computed at runtime.
#     USE THIS for VRAM experiments: watch peak GPU memory drop by ~N×
#     while runtime stays flat (static data — orderings, invIndex, parts —
#     is uploaded once and stays resident across all batches).
#
# --gpu-batch-size S
#     Set an explicit splits-per-batch override.
#     numBatches = ceil(numSplits / S).
#
# --gpu-vram-occupancy-factor F    (default: 0.75)
#     When batching is auto (no explicit --gpu-batches / --gpu-batch-size),
#     use F × freeVRAM as the usable budget for the split batch buffer.
#     The remaining (1-F) is reserved as headroom for driver, kernel stack,
#     and page tables.  E.g. F=0.9 uses 90% of free VRAM per batch.
#
# Memory layout after batching:
#   Permanent on GPU  :  orderings + invIndex + parts   [O(n·k)]
#   Per-batch on GPU  :  splits slice + scores slice     [batchSize × 48 B]
#   Peak VRAM savings :  numSplits / numBatches  (exact reduction)
#
# Examples:

# Disable batching (original behaviour)
./run.sh -i all_gt_bs_rooted_37.tre --gpu --no-gpu-batch -vv

# Auto adaptive batching (default when --gpu and no explicit batch flags)
./run.sh -i all_gt_bs_rooted_37.tre --gpu -vv

# Force exactly 10 batches (VRAM usage drops ~10×; use with nvidia-smi to verify)
./run.sh -i all_gt_bs_rooted_37.tre --gpu --gpu-batches 10 -vv

# Force exactly 1 batch (same as --no-gpu-batch but goes through batch loop)
./run.sh -i all_gt_bs_rooted_37.tre --gpu --gpu-batches 1 -vv

# Explicit batch size of 500 000 splits per launch
./run.sh -i all_gt_bs_rooted_37.tre --gpu --gpu-batch-size 500000 -vv

# Increase VRAM occupancy to 90% (less conservative headroom)
./run.sh -i all_gt_bs_rooted_37.tre --gpu --gpu-vram-occupancy-factor 0.9 -vv

# Conservative 50% occupancy (more headroom for fragmented VRAM situations)
./run.sh -i all_gt_bs_rooted_37.tre --gpu --gpu-vram-occupancy-factor 0.5 -vv

# A10K large run: force 10 batches to reduce peak VRAM on 6M splits
./run-a10k.sh --data-dir data/10k-astral-dataset --tree-type estimated \
    --opts "--search-mode full --gpu --gpu-batches 10"

# Sweep: compare 1, 5, 10 batches to verify runtime is flat and VRAM drops

./run.sh -i all_gt_bs_rooted_37.tre --gpu --gpu-batches 5  -vv
